#!/usr/bin/env python
import sys

data = {}

for line in sys.stdin:
    gender,age = line.strip().split(',')

    if gender not in data:
        data[gender] = [0,0]

    data[gender][0]+= float(age)
    data[gender][1]+= 1

for gender,values in data.items():
    average = values[0] / values[1]
    print("%s = %s" % (gender, average))

**Using pig, find the most occurred start letter in dataset**

**Terminal** 
gedit word1.txt

**Write in file and save** 
1, pooja, R, 20,
2, sam, Thomas, 25
3, priya, S, 20
4, pavan, Rao, 20

**Start HUE,HDFS,YARN**

**Terminal**
pig -x local;


A = LOAD 'word1.txt' USING PigStorage(',')
  AS (id:int, first_name:chararray, last_name:chararray, age:int);
//-- get first letter of first_name
B = FOREACH A GENERATE UPPER(SUBSTRING(first_name, 0, 1)) AS first_letter;
C = GROUP B BY first_letter;
D = FOREACH C GENERATE group AS first_letter, COUNT(B) AS count;
E = ORDER D BY count DESC;
F = LIMIT E 1;
DUMP F;

quit;

from pyspark import SparkContext
from pyspark.sql import SQLContext

# Start spark
sc = SparkContext()
sql = SQLContext(sc)

# Create RDD
rdd = sc.parallelize([
  (1, "AIML", 69),
  (2, "CS", 72),
  (3, "ECE", 99)
])

# Convert RDD to DataFrame
df = sql.createDataFrame(rdd, ["No", "Dept", "No_of_students"])

# Show output on screen
df.show()

sc.stop()

-----------------BASIC COMMANDS--------------------

- mkdir Documents ---> Create directory
- cd Documents ---> change directory
- pwd /home/cloudera/Documents ---> Display current directory
- cd ..  ---> Move to parent directory
- ls ---> list files and directories 
- gedit filename.txt ---> Make new files 
- cat filename.txt ---> display file contents
- cp file1.txt file.txt ---> copy
- mv file2.txt data.txt ---> move or rename
- rm data.txt ---> delete a file 
- rmdir directoryname ---> remove a folder
- rm -r directoryname ---> remove a directory and its content

---------------VIEWING AND SERCHING DATA--------------------

- head ---> Display first lines
- tail ---> Display last lines
- grep ---> Search for text
- wc ---> Count lines, words and characters
- cat ---> Display complete file

-------------SYSTEM AND USER COMMANDS--------------------

- whoami ---> Display current user
- date ---> Display date and time
- history ---> Display previously executed commands
- clear ---> Clear terminal

-------------BASIC HDFS COMMANDS-----------------------

- hdfs dfs -mkdir dirname ---> making a directory
- hdfs dfs -ls
- hdfs dfs -put /home/cloudera/linusfile.txt /user/cloudera/dirname ---> transferring a file from Linux to Hadoop
- hdfs dfs –cat /user/cloudera/dirname/linuxfile.txt ---> view the contents
- hdfs dfs –get /user/cloudera/dirname/linuxfile.txt /home/cloudera/example ---> Hadoop to Linux
- hdfs dfs -touchz /user/cloudera/dirname/test.txt ---> create an empty file inside Hadoop
- hdfs dfs –rm /user/cloudera/dirname/test.txt ---> to remove a file
- hdfs dfs –rm –r /user/cloudera/dirname ---> to remove a directory 
- hdfs dfs -expunge ---> clear trash
- hdfs dfs -count -h /user/cloudera ---> count files & directories
